上下文压缩

1. 概述

当对话过长超出 LLM 上下文窗口时，自动压缩历史消息。
支持多种压缩策略：摘要、滑动窗口、智能选择。

2. Trait 定义

rust
// hermes-core/src/compaction.rs

#[async_trait]
pub trait CompactionStrategy: Send + Sync {
    /// 判断是否需要压缩
    fn should_compact(&self, session: &Session, max_tokens: u32) -> bool;
    
    /// 执行压缩，返回压缩后的消息列表
    async fn compact(&self, session: &Session) -> Result<Vec<Message>>;
    
    fn name(&self) -> &str;
}

3. 压缩策略

3.1 滑动窗口

rust
pub struct SlidingWindowCompaction {
    keep_recent: usize,
    keep_first: usize,  // 保留前几条（通常含系统提示/目标）
}

#[async_trait]
impl CompactionStrategy for SlidingWindowCompaction {
    fn should_compact(&self, session: &Session, max_tokens: u32) -> bool {
        let total_tokens: u32 = session.messages.iter().map(|m| m.estimate_tokens()).sum();
        total_tokens > max_tokens * 80 / 100  // 80% 阈值
    }
    
    async fn compact(&self, session: &Session) -> Result<Vec<Message>> {
        let len = session.messages.len();
        if len <= self.keep_recent + self.keep_first {
            return Ok(session.messages.clone());
        }
        
        let mut result = Vec::new();
        // 保留前 N 条
        result.extend(session.messages[..self.keep_first].iter().cloned());
        // 添加压缩占位符
        result.push(Message::system_text(format!(
            "[Compressed {} messages]", len - self.keep_recent - self.keep_first
        )));
        // 保留后 N 条
        result.extend(session.messages[len - self.keep_recent..].iter().cloned());
        
        Ok(result)
    }
    
    fn name(&self) -> &str { "sliding_window" }
}

3.2 LLM 摘要

rust
pub struct LlmSummaryCompaction {
    provider: Arc<dyn LlmProvider>,
    trigger_tokens: u32,
}

#[async_trait]
impl CompactionStrategy for LlmSummaryCompaction {
    fn should_compact(&self, session: &Session, _max_tokens: u32) -> bool {
        session.messages.len() > 20  // 超过 20 条触发
    }
    
    async fn compact(&self, session: &Session) -> Result<Vec<Message>> {
        // 取前半段生成摘要
        let split = session.messages.len() / 2;
        let to_compact = &session.messages[..split];
        
        let summary_prompt = format!(
            "Summarize the following conversation concisely:\n\n{}",
            to_compact.iter().map(|m| m.text_content()).collect::<Vec<_>>().join("\n")
        );
        
        let response = self.provider.complete(CompletionRequest {
            model: session.meta.model.clone(),
            system: Some("You are a conversation summarizer.".into()),
            messages: vec![Message::user_text(summary_prompt)],
            tools: vec![],
            max_tokens: 2048,
            temperature: Some(0.3),
            enable_caching: false,
        }).await?;
        
        let summary_text = response.text();
        
        // 构建压缩后的消息列表
        let mut result = vec![
            Message::system_text(format!("[Summary of earlier conversation]: {summary_text}"))
        ];
        result.extend(session.messages[split..].iter().cloned());
        
        Ok(result)
    }
    
    fn name(&self) -> &str { "llm_summary" }
}

3.3 智能选择

rust
pub struct SmartCompaction {
    provider: Arc<dyn LlmProvider>,
}

#[async_trait]
impl CompactionStrategy for SmartCompaction {
    async fn compact(&self, session: &Session) -> Result<Vec<Message>> {
        // 保留：第一条（目标）、最近 N 条、所有工具调用结果
        // 压缩：中间的闲聊/探索性对话
        let mut preserved = HashSet::new();
        preserved.insert(0);  // 第一条
        let len = session.messages.len();
        for i in len.saturating_sub(10)..len { preserved.insert(i); }  // 最近 10 条
        for (i, msg) in session.messages.iter().enumerate() {
            if msg.content.iter().any(|c| matches!(c, ContentBlock::ToolResult { .. })) {
                preserved.insert(i);
            }
        }
        
        let mut result = Vec::new();
        let mut compressed_count = 0;
        
        for (i, msg) in session.messages.iter().enumerate() {
            if preserved.contains(&i) {
                result.push(msg.clone());
            } else {
                compressed_count += 1;
            }
        }
        
        if compressed_count > 0 {
            result.push(Message::system_text(format!(
                "[{compressed_count} messages compressed]"
            )));
        }
        
        Ok(result)
    }
}

4. 压缩管理器

rust
pub struct CompactionManager {
    strategies: Vec<Box<dyn CompactionStrategy>>,
    max_tokens: u32,
}

impl CompactionManager {
    pub fn new(max_tokens: u32) -> Self {
        Self {
            strategies: vec![
                Box::new(SlidingWindowCompaction { keep_recent: 10, keep_first: 2 }),
                Box::new(LlmSummaryCompaction::default()),
            ],
            max_tokens,
        }
    }
    
    /// 自动选择策略压缩
    pub async fn auto_compact(&self, session: &Session) -> Result<Vec<Message>> {
        for strategy in &self.strategies {
            if strategy.should_compact(session, self.max_tokens) {
                tracing::info!("Compacting with strategy: {}", strategy.name());
                return strategy.compact(session).await;
            }
        }
        Ok(session.messages.clone())
    }
    
    /// 手动指定策略
    pub async fn compact_with(&self, session: &Session, strategy_name: &str) -> Result<Vec<Message>> {
        let strategy = self.strategies.iter()
            .find(|s| s.name() == strategy_name)
            .ok_or(Error::StrategyNotFound(strategy_name.to_string()))?;
        
        strategy.compact(session).await
    }
}
