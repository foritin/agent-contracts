会话管理

1. 概述

会话管理模块负责 Agent 对话会话的创建、持久化、恢复和归档。
使用 JSONL 格式存储，确保崩溃后可恢复。

2. 数据模型

rust
// hermes-core/src/session.rs

/// 会话元数据（JSONL 第一行）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SessionMeta {
    pub id: String,
    pub created_at: DateTime<Utc>,
    pub model: String,
    pub provider: String,
    pub title: Option<String>,
}

/// JSONL 中的每一行
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum SessionEvent {
    Meta(SessionMeta),
    Message(Message),
    Usage(Usage),
    ToolCall { name: String, input: Value },
    ToolResult { call_id: String, output: Value },
    System { event: String, data: Value },
}

/// 内存中的会话视图
#[derive(Debug, Clone)]
pub struct Session {
    pub meta: SessionMeta,
    pub messages: Vec<Message>,
    pub total_input_tokens: u32,
    pub total_output_tokens: u32,
    pub total_tool_calls: u32,
    pub status: SessionStatus,
}

pub enum SessionStatus {
    Active,
    Paused,
    Completed,
    Archived,
    Error(String),
}

3. 持久化层

rust
// hermes-store/src/session_store.rs

pub struct SessionStore {
    base_dir: PathBuf,
}

impl SessionStore {
    pub fn new(base_dir: PathBuf) -> Self {
        Self { base_dir }
    }
    
    /// 创建新会话
    pub async fn create(&self, model: &str, provider: &str) -> Result<Session> {
        let meta = SessionMeta::new(model, provider);
        let path = self.session_path(&meta.id);
        
        // 写入 meta 行
        let mut file = OpenOptions::new()
            .create(true)
            .append(true)
            .open(&path)?;
        
        let meta_line = serde_json::to_string(&SessionEvent::Meta(meta.clone()))?;
        writeln!(file, "{}", meta_line)?;
        
        Ok(Session::new(meta))
    }
    
    /// 追加事件到会话日志
    pub async fn append(&self, session_id: &str, event: SessionEvent) -> Result<()> {
        let path = self.session_path(session_id);
        let mut file = OpenOptions::new().append(true).open(&path)?;
        let line = serde_json::to_string(&event)?;
        writeln!(file, "{}", line)?;
        Ok(())
    }
    
    /// 读取完整会话
    pub async fn load(&self, session_id: &str) -> Result<Session> {
        let path = self.session_path(session_id);
        let file = File::open(&path)?;
        let reader = BufReader::new(file);
        
        let mut meta = None;
        let mut messages = Vec::new();
        let mut usage = Usage::default();
        
        for line in reader.lines() {
            let line = line?;
            let event: SessionEvent = serde_json::from_str(&line)?;
            
            match event {
                SessionEvent::Meta(m) => meta = Some(m),
                SessionEvent::Message(msg) => messages.push(msg),
                SessionEvent::Usage(u) => usage += u,
                _ => {}
            }
        }
        
        Ok(Session {
            meta: meta.ok_or(Error::InvalidSession)?,
            messages,
            total_input_tokens: usage.input_tokens,
            total_output_tokens: usage.output_tokens,
        })
    }
    
    /// 列出所有会话
    pub async fn list(&self) -> Result<Vec<SessionMeta>> {
        let mut sessions = Vec::new();
        for entry in std::fs::read_dir(&self.base_dir)? {
            let path = entry?.path();
            if path.extension().and_then(|e| e.to_str()) == Some("jsonl") {
                if let Ok(meta) = self.read_meta(&path).await {
                    sessions.push(meta);
                }
            }
        }
        sessions.sort_by(|a, b| b.created_at.cmp(&a.created_at));
        Ok(sessions)
    }
    
    fn session_path(&self, id: &str) -> PathBuf {
        self.base_dir.join(format!("{}.jsonl", id))
    }
}

4. 崩溃恢复

rust
impl SessionStore {
    /// 恢复中断的会话
    pub async fn recover(&self, session_id: &str) -> Result<Session> {
        let mut session = self.load(session_id).await?;
        
        // 检查最后一行是否完整
        let path = self.session_path(session_id);
        let content = tokio::fs::read_to_string(&path).await?;
        
        if content.ends_with('\n') {
            // 完整，直接返回
            Ok(session)
        } else {
            // 截断最后一行
            let mut lines: Vec<&str> = content.lines().collect();
            lines.pop();  // 移除不完整的行
            
            let recovered = lines.join("\n") + "\n";
            tokio::fs::write(&path, recovered).await?;
            
            // 重新加载
            self.load(session_id).await
        }
    }
}

5. 会话归档

rust
impl SessionStore {
    /// 归档会话（压缩 + 标记）
    pub async fn archive(&self, session_id: &str) -> Result<()> {
        let path = self.session_path(session_id);
        let archive_path = self.base_dir.join(format!("{}.jsonl.gz", session_id));
        
        // gzip 压缩
        let input = File::open(&path)?;
        let output = File::create(&archive_path)?;
        let mut encoder = GzBuilder::new()
            .write(output, Compression::default());
        
        io::copy(&mut &input, &mut encoder)?;
        encoder.finish()?;
        
        // 删除原文件
        std::fs::remove_file(&path)?;
        
        Ok(())
    }
}

6. 使用示例

rust
use hermes_store::SessionStore;
use hermes_core::{SessionEvent, Message};

#[tokio::main]
async fn main() -> Result<()> {
    let store = SessionStore::new(PathBuf::from("./sessions"));
    
    // 创建会话
    let session = store.create("claude-sonnet-4", "anthropic").await?;
    println!("Created session: {}", session.meta.id);
    
    // 追加消息
    store.append(&session.meta.id, SessionEvent::Message(
        Message::user_text("Hello!")
    )).await?;
    
    // 加载会话
    let loaded = store.load(&session.meta.id).await?;
    println!("Messages: {}", loaded.messages.len());
    
    Ok(())
}
