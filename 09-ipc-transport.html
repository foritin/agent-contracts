IPC 传输层

1. 概述

Unix Socket（macOS/Linux）和 Named Pipe（Windows）的 JSON-RPC 2.0 传输实现。
用于 Charter Main ↔ Agent Worker 的进程间通信。

2. 协议定义

jsonc
// 请求
{
  "jsonrpc": "2.0",
  "id": "req-1",
  "method": "tool.execute",
  "params": { "name": "read_file", "input": { "path": "/foo.txt" } }
}

// 响应
{
  "jsonrpc": "2.0",
  "id": "req-1",
  "result": { "content": "Hello World" }
}

// 错误
{
  "jsonrpc": "2.0",
  "id": "req-1",
  "error": { "code": -32603, "message": "tool not found" }
}

// 事件（无 id）
{
  "jsonrpc": "2.0",
  "method": "agent.event",
  "params": { "type": "message", "text": "Hello" }
}

3. 传输帧格式

text
[4 bytes: payload length (big-endian u32)][N bytes: JSON payload]

4. 服务端实现

rust
// hermes-ipc/src/server.rs

pub struct IpcServer {
    listener: UnixListener,
    handlers: HashMap<String, Box<dyn IpcHandler>>,
}

#[async_trait]
trait IpcHandler: Send + Sync {
    async fn handle(&self, params: Value) -> Result<Value>;
}

impl IpcServer {
    pub fn new(socket_path: PathBuf) -> Result<Self> {
        // 清理旧 socket
        if socket_path.exists() {
            std::fs::remove_file(&socket_path)?;
        }
        
        let listener = UnixListener::bind(&socket_path)?;
        
        // 设置权限（仅 owner）
        #[cfg(unix)]
        {
            use std::os::unix::fs::PermissionsExt;
            std::fs::set_permissions(&socket_path, Permissions::from_mode(0o600))?;
        }
        
        Ok(Self {
            listener,
            handlers: HashMap::new(),
        })
    }
    
    pub fn register(&mut self, method: &str, handler: Box<dyn IpcHandler>) {
        self.handlers.insert(method.to_string(), handler);
    }
    
    pub async fn serve(&self) -> Result<()> {
        loop {
            let (mut stream, _addr) = self.listener.accept().await?;
            
            let handlers = self.handlers.clone();
            tokio::spawn(async move {
                if let Err(e) = Self::handle_connection(stream, handlers).await {
                    tracing::error!("IPC connection error: {}", e);
                }
            });
        }
    }
    
    async fn handle_connection(
        mut stream: UnixStream,
        handlers: HashMap<String, Box<dyn IpcHandler>>,
    ) -> Result<()> {
        loop {
            // 读取帧头
            let mut len_buf = [0u8; 4];
            stream.read_exact(&mut len_buf).await?;
            let len = u32::from_be_bytes(len_buf) as usize;
            
            // 读取 payload
            let mut buf = vec![0u8; len];
            stream.read_exact(&mut buf).await?;
            
            let request: JsonRpcRequest = serde_json::from_slice(&buf)?;
            
            // 处理
            let response = match handlers.get(&request.method) {
                Some(handler) => {
                    match handler.handle(request.params.unwrap_or(Value::Null)).await {
                        Ok(result) => JsonRpcResponse {
                            jsonrpc: "2.0".into(),
                            id: request.id,
                            result: Some(result),
                            error: None,
                        },
                        Err(e) => JsonRpcResponse {
                            jsonrpc: "2.0".into(),
                            id: request.id,
                            result: None,
                            error: Some(JsonRpcError {
                                code: -32603,
                                message: e.to_string(),
                            }),
                        },
                    }
                }
                None => JsonRpcResponse {
                    jsonrpc: "2.0".into(),
                    id: request.id,
                    result: None,
                    error: Some(JsonRpcError {
                        code: -32601,
                        message: format!("method not found: {}", request.method),
                    }),
                },
            };
            
            // 发送响应
            let response_json = serde_json::to_vec(&response)?;
            let response_len = response_json.len() as u32;
            stream.write_all(&response_len.to_be_bytes()).await?;
            stream.write_all(&response_json).await?;
        }
    }
}

5. 客户端实现

rust
// hermes-ipc/src/client.rs

pub struct IpcClient {
    stream: UnixStream,
    pending: Arc<Mutex<HashMap<String, oneshot::Sender<JsonRpcResponse>>>>,
    next_id: Arc<AtomicU64>,
}

impl IpcClient {
    pub async fn connect(socket_path: &Path) -> Result<Self> {
        let stream = UnixStream::connect(socket_path).await?;
        
        let client = Self {
            stream,
            pending: Arc::new(Mutex::new(HashMap::new())),
            next_id: Arc::new(AtomicU64::new(1)),
        };
        
        // 启动接收循环
        client.start_recv_loop();
        
        Ok(client)
    }
    
    pub async fn call(&self, method: &str, params: Value) -> Result<Value> {
        let id = self.next_id.fetch_add(1, Ordering::SeqCst).to_string();
        let (tx, rx) = oneshot::channel();
        
        self.pending.lock().await.insert(id.clone(), tx);
        
        let request = JsonRpcRequest {
            jsonrpc: "2.0".into(),
            id: Some(id),
            method: method.to_string(),
            params: Some(params),
        };
        
        let json = serde_json::to_vec(&request)?;
        let len = json.len() as u32;
        self.stream.write_all(&len.to_be_bytes()).await?;
        self.stream.write_all(&json).await?;
        
        let response = rx.await.map_err(|_| Error::IpcTimeout)?;
        
        match response.error {
            Some(e) => Err(Error::Ipc(e.message)),
            None => Ok(response.result.unwrap_or(Value::Null)),
        }
    }
    
    fn start_recv_loop(&self) {
        let pending = self.pending.clone();
        let mut stream = self.stream.try_clone().unwrap();
        
        tokio::spawn(async move {
            loop {
                // 读取并分发响应
            }
        });
    }
}

6. Windows Named Pipe

rust
#[cfg(windows)]
pub use windows::{IpcServerNamedPipe, IpcClientNamedPipe};

#[cfg(windows)]
mod windows {
    use tokio::net::windows::named_pipe::{ServerOptions, ClientOptions};
    
    pub struct IpcServerNamedPipe {
        pipe: tokio::net::windows::named_pipe::NamedPipeServer,
    }
    
    impl IpcServerNamedPipe {
        pub fn new(name: &str) -> Result<Self> {
            let pipe = ServerOptions::new()
                .first_pipe_instance(true)
                .create(name)?;
            Ok(Self { pipe })
        }
    }
}
