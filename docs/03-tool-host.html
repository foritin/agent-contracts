ToolHost 抽象

1. 概述

ToolHost 是工具的执行抽象层。它聚合多个工具来源（MCP server、内置工具、网关代理），
为 Agent 循环提供统一的工具调用接口。

2. Trait 定义

rust
// hermes-core/src/tool_host.rs

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ToolCallOutcome {
    pub content: String,
    pub is_error: bool,
    pub metadata: Option<Value>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ToolSpec {
    pub name: String,
    pub description: String,
    pub input_schema: Value,
    pub source: ToolSource,
    pub requires_confirmation: bool,
}

pub enum ToolSource {
    Builtin,           // 内置工具
    MCP { server: String },  // MCP 工具
    Custom { id: String },   // 产品层自定义工具（如网关代理等）
}

#[async_trait]
pub trait ToolHost: Send + Sync {
    /// 列出所有可用工具
    async fn list_tools(&self) -> Result<Vec<ToolSpec>>;
    
    /// 执行工具调用
    async fn call(&self, name: &str, args: Value) -> Result<ToolCallOutcome>;
    
    /// 批量执行（用于并行工具调用）
    async fn call_batch(&self, calls: Vec<(String, Value)>) -> Vec<Result<ToolCallOutcome>> {
        let mut results = Vec::with_capacity(calls.len());
        for (name, args) in calls {
            results.push(self.call(&name, args).await);
        }
        results
    }
}

rust
pub struct NullToolHost;

#[async_trait]
impl ToolHost for NullToolHost {
    async fn list_tools(&self) -> Result<Vec<ToolSpec>> {
        Ok(Vec::new())
    }
    
    async fn call(&self, name: &str, _: Value) -> Result<ToolCallOutcome> {
        Err(Error::ToolHost(format!("no tool host configured; cannot call {name:?}")))
    }
}

4. MCP 工具聚合

rust
// hermes-mcp/src/host.rs

pub struct McpToolHost {
    servers: Vec<McpServer>,
    cache: RwLock<HashMap<String, Vec<ToolSpec>>>,
}

impl McpToolHost {
    pub fn new(servers: Vec<McpServer>) -> Self {
        Self {
            servers,
            cache: RwLock::new(HashMap::new()),
        }
    }
    
    pub async fn add_server(&mut self, server: McpServer) -> Result<()> {
        server.connect().await?;
        self.servers.push(server);
        self.cache.write().await.clear();  // 失效缓存
        Ok(())
    }
}

#[async_trait]
impl ToolHost for McpToolHost {
    async fn list_tools(&self) -> Result<Vec<ToolSpec>> {
        // 检查缓存
        let cache = self.cache.read().await;
        let cached: Vec<ToolSpec> = cache.values().flatten().cloned().collect();
        if !cached.is_empty() {
            return Ok(cached);
        }
        drop(cache);
        
        // 从所有 server 收集
        let mut all_tools = Vec::new();
        let mut cache = self.cache.write().await;
        
        for server in &self.servers {
            let tools = server.list_tools().await?;
            let specs: Vec<ToolSpec> = tools.into_iter().map(|t| ToolSpec {
                name.name: format!("{}__{}", server.name(), t.name),
                description: t.description,
                input_schema: t.input_schema,
                source: ToolSource::MCP { server: server.name().to_string() },
                requires_confirmation: true,  // MCP 工具默认需要确认
            }).collect();
            
            cache.insert(server.name().to_string(), specs.clone());
            all_tools.extend(specs);
        }
        
        Ok(all_tools)
    }
    
    async fn call(&self, name: &str, args: Value) -> Result<ToolCallOutcome> {
        // 解析 namespaced name: "<server>__<tool>"
        let parts: Vec<&str> = name.splitn(2, "__").collect();
        if parts.len() != 2 {
            return Err(Error::ToolHost(format!("invalid tool name format: {name:?}")));
        }
        
        let (server_name, tool_name) = (parts[0], parts[1]);
        
        // 找到对应 server
        let server = self.servers.iter()
            .find(|s| s.name() == server_name)
            .ok_or(Error::ToolHost(format!("server not found: {server_name}")))?;
        
        server.call_tool(tool_name, args).await
    }
}

5. 产品层 ToolHost 桥接（示例，不在公共 crate）

本节为产品层实现示例，不属于 agent-core 公共层。产品（如桌面 IDE）可将 ToolHost 桥接到其专有网关；公共层只提供 ToolHost trait 与 ToolSource::Custom 钩子，不定义产品专属网关类型。

rust
// 产品层示例：将 ToolHost 桥接到产品专有网关

pub struct GatewayToolHost {
    gateway: Arc<ToolGateway>,
    task_id: String,
    terminal_id: Option<String>,
}

impl GatewayToolHost {
    pub fn new(gateway: Arc<ToolGateway>, task_id: String) -> Self {
        Self { gateway, task_id, terminal_id: None }
    }
}

#[async_trait]
impl ToolHost for GatewayToolHost {
    async fn list_tools(&self) -> Result<Vec<ToolSpec>> {
        let tools = self.gateway.list_tools(&self.task_id).await?;
        Ok(tools.into_iter().map(|t| ToolSpec {
            name: t.name,
            description: t.description,
            input_schema: t.input_schema,
            source: ToolSource::Custom { id: "gateway".into() },
            requires_confirmation: t.risk_level >= RiskLevel::R2,
        }).collect())
    }
    
    async fn call(&self, name: &str, args: Value) -> Result<ToolCallOutcome> {
        let request = ToolCallRequest {
            tool_name: name.to_string(),
            input: args,
            task_id: self.task_id.clone(),
            caller_terminal: self.terminal_id.clone(),
        };
        
        match self.gateway.execute(request).await {
            Ok(result) => Ok(ToolCallOutcome {
                content: serde_json::to_string(&result.output)?,
                is_error: false,
                metadata: Some(json!({
                    "risk_level": result.risk_level,
                    "tool_call_id": result.call_id,
                })),
            }),
            Err(e) => Ok(ToolCallOutcome {
                content: format!("Tool call failed: {e}"),
                is_error: true,
                metadata: None,
            }),
        }
    }
}

6. 组合 ToolHost

rust
/// 组合多个 ToolHost，按优先级查找
pub struct CompositeToolHost {
    hosts: Vec<Box<dyn ToolHost>>,
}

impl CompositeToolHost {
    pub fn new() -> Self {
        Self { hosts: Vec::new() }
    }
    
    pub fn add_host(&mut self, host: Box<dyn ToolHost>) {
        self.hosts.push(host);
    }
}

#[async_trait]
impl ToolHost for CompositeToolHost {
    async fn list_tools(&self) -> Result<Vec<ToolSpec>> {
        let mut all_tools = Vec::new();
        for host in &self.hosts {
            all_tools.extend(host.list_tools().await?);
        }
        Ok(all_tools)
    }
    
    async fn call(&self, name: &str, args: Value) -> Result<ToolCallOutcome> {
        // 按优先级尝试每个 host
        for host in &self.hosts {
            if host.list_tools().await?.iter().any(|t| t.name == name) {
                return host.call(name, args).await;
            }
        }
        Err(Error::ToolHost(format!("tool not found: {name:?}")))
    }
}
