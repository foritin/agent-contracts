LLM Provider 抽象

1. 概述

定义统一的 LLM Provider trait，支持多模型后端（Anthropic Claude、DeepSeek、OpenAI 兼容 API）。

2. Trait 定义

rust
// hermes-core/src/provider.rs

#[async_trait]
pub trait LlmProvider: Send + Sync {
    async fn complete(
        &self,
        request: CompletionRequest,
    ) -> Result<CompletionResponse>;
    
    async fn stream(
        &self,
        request: CompletionRequest,
    ) -> Result<BoxStream<'static, StreamEvent>>;
    
    fn capabilities(&self) -> Capabilities;
    fn name(&self) -> &str;
}

3. 请求/响应类型

rust
pub struct CompletionRequest {
    pub model: String,
    pub system: Option<String>,
    pub messages: Vec<Message>,
    pub tools: Vec<ToolSpec>,
    pub max_tokens: u32,
    pub temperature: Option<f32>,
    pub enable_caching: bool,
}

pub struct CompletionResponse {
    pub content: Vec<ContentBlock>,
    pub stop_reason: StopReason,
    pub usage: Usage,
}

pub enum StreamEvent {
    TextDelta { text: String },
    ToolUseStart { id: String, name: String },
    ToolUseDelta { id: String, input_json: String },
    ToolUseComplete { id: String, input: Value },
    Stop { reason: StopReason },
    Usage(Usage),
}

pub enum StopReason {
    EndTurn,
    ToolUse,
    MaxTokens,
    StopSequence,
    Other(String),
}

pub struct Usage {
    pub input_tokens: u32,
    pub output_tokens: u32,
    pub cache_read_tokens: Option<u32>,
    pub cache_write_tokens: Option<u32>,
}

5. Provider 工厂

rust
// hermes-llm/src/lib.rs

pub enum ProviderConfig {
    Anthropic {
        api_key: String,
        model: String,
        base_url: Option<String>,
    },
    OpenAi {
        api_key: String,
        model: String,
        base_url: String,
    },
    DeepSeek {
        api_key: String,
        model: String,
    },
}

pub fn create_provider(config: ProviderConfig) -> Result<Box<dyn LlmProvider>> {
    match config {
        ProviderConfig::Anthropic { api_key, model, base_url } => {
            Ok(Box::new(AnthropicProvider::new(api_key, model, base_url)))
        }
        ProviderConfig::OpenAi { api_key, model, base_url } => {
            Ok(Box::new(OpenAiProvider::new(api_key, model, base_url)))
        }
        ProviderConfig::DeepSeek { api_key, model } => {
            Ok(Box::new(DeepSeekProvider::new(api_key, model)))
        }
    }
}

6. 能力协商

rust
pub struct Capabilities {
    pub supports_streaming: bool,
    pub supports_tool_use: bool,
    pub supports_vision: bool,
    pub supports_prompt_caching: bool,
    pub max_context_tokens: u32,
}

impl Capabilities {
    fn can_use_tools(&self) -> bool {
        self.supports_tool_use
    }
    
    fn cache_key(&self) -> &'static str {
        if self.supports_prompt_caching {
            "cached"
        } else {
            "uncached"
        }
    }
}

7. Prompt Caching

rust
// Anthropic prompt caching 实现
fn apply_prompt_caching(&self, request: &mut AnthropicRequest) {
    // 对 system + tools + 前几条 user turn 添加 cache_control
    if let Some(ref mut system) = request.system {
        system.cache_control = Some(CacheControl { type_: "ephemeral" });
    }
}

8. 使用示例

rust
use hermes_llm::{ProviderConfig, create_provider};
use hermes_core::{CompletionRequest, Message};

#[tokio::main]
async fn main() -> Result<()> {
    let provider = create_provider(ProviderConfig::Anthropic {
        api_key: std::env::var("ANTHROPIC_API_KEY")?,
        model: "claude-sonnet-4-20250514".into(),
        base_url: None,
    })?;
    
    let request = CompletionRequest {
        model: "claude-sonnet-4-20250514".into(),
        system: Some("You are a helpful assistant.".into()),
        messages: vec![Message::user_text("Hello!")],
        tools: vec![],
        max_tokens: 1024,
        temperature: None,
        enable_caching: true,
    };
    
    let response = provider.complete(request).await?;
    println!("Response: {:?}", response.content);
    
    Ok(())
}
