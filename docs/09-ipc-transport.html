IPC 传输层

1. 概述

Unix Socket（macOS/Linux）和 Named Pipe（Windows）的 JSON-RPC 2.0 传输实现。
用于 R-Code Main ↔ Agent Worker 的进程间通信。

2. 协议定义

jsonc
// 请求
{
  "jsonrpc": "2.0",
  "id": "req-1",
  "method": "tool.execute",
  "params": { "name": "read_file", "input": { "path": "/foo.txt" } }
}

// 响应
{
  "jsonrpc": "2.0",
  "id": "req-1",
  "result": { "content": "Hello World" }
}

// 错误
{
  "jsonrpc": "2.0",
  "id": "req-1",
  "error": { "code": -32603, "message": "tool not found" }
}

// 事件（无 id）
{
  "jsonrpc": "2.0",
  "method": "agent.event",
  "params": { "type": "message", "text": "Hello" }
}

3. 传输帧格式

text
[4 bytes: payload length (big-endian u32)][N bytes: JSON payload]

4. 服务端实现

rust
// agent-ipc/src/server.rs

use std::path::PathBuf;
use std::sync::Arc;
use tokio::io::{AsyncRead, AsyncWrite};

// 平台特定的 listener
#[cfg(unix)]
type PlatformListener = tokio::net::UnixListener;

#[cfg(windows)]
struct WindowsPipeListener {
    pipe_name: String,
}

/// 请求处理器 trait（跨平台）。
#[async_trait::async_trait]
pub trait IpcHandler: Send + Sync {
    async fn handle(&self, params: Value) -> Result<Value>;
}

/// IPC 服务端（跨平台）。
pub struct IpcServer {
    #[cfg(unix)]
    listener: PlatformListener,
    #[cfg(windows)]
    listener: WindowsPipeListener,
    handlers: HashMap<String, Arc<dyn IpcHandler>>,
    socket_path: PathBuf,
}

impl IpcServer {
    /// 绑定 IPC 端点。
    /// - Unix: 绑定 Unix Socket，清理旧文件，设置 0o600 权限。
    /// - Windows: 记录 Named Pipe 名称（实例在 serve() 中按需创建）。
    pub fn bind(socket_path: PathBuf) -> Result<Self> {
        #[cfg(unix)]
        {
            // 清理旧 socket
            if socket_path.exists() {
                std::fs::remove_file(&socket_path)?;
            }
            let listener = tokio::net::UnixListener::bind(&socket_path)?;

            // 设置权限（仅 owner）
            use std::os::unix::fs::PermissionsExt;
            std::fs::set_permissions(
                &socket_path,
                std::fs::Permissions::from_mode(0o600),
            )?;

            Ok(Self { listener, handlers: HashMap::new(), socket_path })
        }

        #[cfg(windows)]
        {
            // 路径转换: /tmp/r-code.sock -> \\.\pipe\r-code-r-code.sock
            let pipe_name = format!(
                r"\\.\pipe\r-code-{}",
                socket_path.file_name().and_then(|n| n.to_str()).unwrap_or("default")
            );
            Ok(Self {
                listener: WindowsPipeListener { pipe_name },
                handlers: HashMap::new(),
                socket_path,
            })
        }
    }

    pub fn register(&mut self, method: &str, handler: Arc<dyn IpcHandler>) {
        self.handlers.insert(method.to_string(), handler);
    }

    pub async fn serve(self) -> Result<()> {
        #[cfg(unix)]
        {
            loop {
                let (stream, _addr) = self.listener.accept().await?;
                let handlers = self.handlers.clone();
                tokio::spawn(async move {
                    if let Err(e) = Self::handle_connection(stream, handlers).await {
                        tracing::error!("IPC connection error: {e}");
                    }
                });
            }
        }

        #[cfg(windows)]
        {
            use tokio::net::windows::named_pipe::ServerOptions;
            let pipe_name = self.listener.pipe_name.clone();
            loop {
                // 每个连接创建独立的 pipe 实例
                let server = ServerOptions::new()
                    .first_pipe_instance(false)
                    .create(&pipe_name)?;
                server.connect().await?;

                let handlers = self.handlers.clone();
                tokio::spawn(async move {
                    if let Err(e) = Self::handle_connection(server, handlers).await {
                        tracing::error!("IPC connection error: {e}");
                    }
                });
            }
        }
    }

    /// 处理单个客户端连接（泛型，跨平台复用）。
    async fn handle_connection<S>(
        mut stream: S,
        handlers: HashMap<String, Arc<dyn IpcHandler>>,
    ) -> Result<()>
    where
        S: AsyncRead + AsyncWrite + Unpin,
    {
        loop {
            let request: JsonRpcRequest = read_message(&mut stream).await?;
            let response = match handlers.get(&request.method) {
                Some(h) => match h.handle(request.params.unwrap_or(Value::Null)).await {
                    Ok(result) => JsonRpcResponse::success(
                        request.id.unwrap_or_default(),
                        result,
                    ),
                    Err(e) => JsonRpcResponse::error(
                        request.id,
                        JsonRpcError::INTERNAL_ERROR,
                        e.to_string(),
                    ),
                },
                None => JsonRpcResponse::error(
                    request.id,
                    JsonRpcError::METHOD_NOT_FOUND,
                    format!("method not found: {}", request.method),
                ),
            };
            if request.id.is_some() {
                write_message(&mut stream, &response).await?;
                stream.flush().await;
            }
        }
    }
}

5. 客户端实现

rust
// agent-ipc/src/client.rs

use std::sync::atomic::{AtomicU64, Ordering};
use std::sync::Arc;
use tokio::io::{AsyncRead, AsyncWrite};
use tokio::sync::{oneshot, Mutex};

/// Named Pipe 名称前缀（Windows）。
#[cfg(windows)]
const PIPE_PREFIX: &str = r"\\.\pipe\r-code-";

/// IPC 客户端（跨平台）。
///
/// 内部用 boxed trait object 持有 read/write 半边：
/// - Unix: OwnedReadHalf / OwnedWriteHalf（UnixStream::into_split）
/// - Windows: ReadHalf / WriteHalf（tokio::io::split）
pub struct IpcClient {
    write: Arc<Mutex<Box<dyn AsyncWrite + Unpin + Send>>>,
    pending: Arc<Mutex<HashMap<String, oneshot::Sender<JsonRpcResponse>>>>,
    next_id: Arc<AtomicU64>,
}

impl IpcClient {
    /// 连接到 IPC 服务端。
    /// - Unix: UnixStream::connect + into_split
    /// - Windows: ClientOptions::open + tokio::io::split
    pub async fn connect(path: &Path) -> Result<Self> {
        #[cfg(unix)]
        {
            let stream = tokio::net::UnixStream::connect(path).await?;
            let (read, write) = stream.into_split();
            Ok(Self::from_halves(Box::new(read), Box::new(write)))
        }

        #[cfg(windows)]
        {
            // 路径转换: /tmp/r-code.sock -> \\.\pipe\r-code-r-code.sock
            let pipe_name = if path.to_string_lossy().starts_with(PIPE_PREFIX) {
                path.to_string_lossy().to_string()
            } else {
                format!(
                    "{}{}",
                    PIPE_PREFIX,
                    path.file_name().and_then(|n| n.to_str()).unwrap_or("default")
                )
            };
            let stream = tokio::net::windows::named_pipe::ClientOptions::new()
                .open(&pipe_name)?;
            let (read, write) = tokio::io::split(stream);
            Ok(Self::from_halves(Box::new(read), Box::new(write)))
        }

        #[cfg(not(any(unix, windows)))]
        {
            let _ = path;
            Err(Error::Ipc("unsupported platform".into()))
        }
    }

    /// 从已拆分的 read/write 半边构造客户端。
    fn from_halves(
        read: Box<dyn AsyncRead + Unpin + Send>,
        write: Box<dyn AsyncWrite + Unpin + Send>,
    ) -> Self {
        let pending = Arc::new(Mutex::new(HashMap::new()));
        let pending_clone = pending.clone();
        tokio::spawn(async move {
            Self::recv_loop(read, pending_clone).await;
        });
        Self {
            write: Arc::new(Mutex::new(write)),
            pending,
            next_id: Arc::new(AtomicU64::new(1)),
        }
    }

    pub async fn call(&self, method: &str, params: Value) -> Result<Value> {
        let id = self.next_id.fetch_add(1, Ordering::SeqCst).to_string();
        let (tx, rx) = oneshot::channel();
        self.pending.lock().await.insert(id.clone(), tx);

        let request = JsonRpcRequest::new(method, id.clone(), Some(params));
        {
            let mut w = self.write.lock().await;
            write_message(&mut *w, &request).await?;
        }

        let response = rx.await.map_err(|_| Error::IpcTimeout)?;
        match response.error {
            Some(e) => Err(Error::Ipc(format!("[{}] {}", e.code, e.message))),
            None => Ok(response.result.unwrap_or(Value::Null)),
        }
    }

    async fn recv_loop(
        mut read: Box<dyn AsyncRead + Unpin + Send>,
        pending: Arc<Mutex<HashMap<String, oneshot::Sender<JsonRpcResponse>>>>,
    ) {
        loop {
            match read_message::<_, JsonRpcResponse>(&mut read).await {
                Ok(resp) => {
                    if let Some(id) = resp.id.clone() {
                        if let Some(tx) = pending.lock().await.remove(&id) {
                            let _ = tx.send(resp);
                        }
                    }
                }
                Err(_) => break,
            }
        }
    }
}

6. Windows Named Pipe 适配

Windows 不支持 Unix Socket，改用 Named Pipe。由于 tokio 的 NamedPipeServer / NamedPipeClient 同样实现 AsyncRead + AsyncWrite，服务端 handle_connection<S> 与客户端的 boxed 半边无需任何平台分支即可复用，差异仅集中在端点绑定 / 连接处。

rust
// 路径转换: /tmp/r-code.sock -> \\.\pipe\r-code-r-code.sock
//
// 服务端 (bind): WindowsPipeListener { pipe_name } 仅记录名称，
//   实际 pipe 实例在 serve() 中按 ServerOptions::new().create() 创建。
// 客户端 (connect): ClientOptions::new().open(&pipe_name) 直连。

#[cfg(windows)]
const PIPE_PREFIX: &str = r"\\.\pipe\r-code-";

fn to_pipe_name(path: &Path) -> String {
    let s = path.to_string_lossy();
    if s.starts_with(PIPE_PREFIX) {
        s.into_owned()
    } else {
        format!(
            "{}{}",
            PIPE_PREFIX,
            path.file_name().and_then(|n| n.to_str()).unwrap_or("default")
        )
    }
}
