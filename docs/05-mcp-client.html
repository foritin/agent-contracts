MCP 客户端

1. 概述

MCP（Model Context Protocol）客户端模块，封装 rmcp crate，
支持 stdio 子进程和 Streamable HTTP 两种传输方式。

2. 配置格式

toml
# config.toml
[mcp_servers.filesystem]
command = "npx"
args = ["-y", "@modelcontextprotocol/server-filesystem", "/workspace"]
env = { "DEBUG" = "false" }

[mcp_servers.github]
type = "http"
url = "https://api.github.com/mcp"
Authorization = "Bearer ${GITHUB_TOKEN}"

3. 配置解析

rust
// agent-mcp/src/config.rs

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "type", rename_all = "snake_case")]
pub enum ServerSpec {
    Stdio {
        command: String,
        args: Vec<String>,
        env: HashMap<String, String>,
    },
    Http {
        url: String,
        headers: HashMap<String, String>,
    },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct McpConfig {
    pub servers: HashMap<String, ServerSpec>,
}

impl McpConfig {
    pub fn load(path: &Path) -> Result<Self> {
        let content = std::fs::read_to_string(path)?;
        let config: McpConfig = toml::from_str(&content)?;
        Ok(config)
    }
    
    pub fn load_default() -> Result<Self> {
        let path = dirs::home_dir()
            .ok_or(Error::Config("home dir not found"))?
            .join(".hermes/config.toml");
        Self::load(&path)
    }
}

4. MCP Server 连接

rust
// agent-mcp/src/server.rs

pub struct McpServer {
    name: String,
    spec: ServerSpec,
    client: Option<rmcp::Client>,
    tools: Vec<Tool>,
}

impl McpServer {
    pub fn new(name: String, spec: ServerSpec) -> Self {
        Self { name, spec, client: None, tools: Vec::new() }
    }
    
    pub async fn connect(&mut self) -> Result<()> {
        let transport = match &self.spec {
            ServerSpec::Stdio { command, args, env } => {
                rmcp::transport::child_process::ChildProcessTransport::new(
                    command, args, env.clone()
                )?
            }
            ServerSpec::Http { url, headers } => {
                rmcp::transport::http::HttpTransport::new(url, headers.clone())?
            }
        };
        
        let client = rmcp::Client::new(transport);
        client.initialize(InitializeRequest {
            name: "hermes".to_string(),
            version: env!("CARGO_PKG_VERSION").to_string(),
            capabilities: ClientCapabilities::default(),
        }).await?;
        
        self.client = Some(client);
        self.refresh_tools().await?;
        Ok(())
    }
    
    pub async fn refresh_tools(&mut self) -> Result<()> {
        let client = self.client.as_ref().ok_or(Error::NotConnected)?;
        let response = client.list_tools(ListToolsRequest::default()).await?;
        self.tools = response.tools;
        Ok(())
    }
    
    pub async fn call_tool(&self, name: &str, args: Value) -> Result<ToolCallOutcome> {
        let client = self.client.as_ref().ok_or(Error::NotConnected)?;
        let response = client.call_tool(CallToolRequest {
            name: name.to_string(),
            arguments: args,
        }).await?;
        
        Ok(ToolCallOutcome {
            content: response.content.first()
                .map(|c| c.to_string())
                .unwrap_or_default(),
            is_error: response.is_error,
            metadata: response.metadata,
        })
    }
}

5. MCP Host（聚合层）

rust
// agent-mcp/src/host.rs

pub struct McpToolHost {
    servers: Vec<McpServer>,
    cache: RwLock<HashMap<String, Vec<ToolSpec>>>,
}

impl McpToolHost {
    pub fn from_config(config: McpConfig) -> Result<Self> {
        let servers: Vec<McpServer> = config.servers.into_iter()
            .map(|(name, spec)| McpServer::new(name, spec))
            .collect();
        
        Ok(Self {
            servers,
            cache: RwLock::new(HashMap::new()),
        })
    }
    
    pub async fn connect_all(&mut self) -> Result<()> {
        for server in &mut self.servers {
            if let Err(e) = server.connect().await {
                tracing::warn!("Failed to connect MCP server '{}': {}", server.name, e);
            }
        }
        Ok(())
    }
}

#[async_trait]
impl ToolHost for McpToolHost {
    async fn list_tools(&self) -> Result<Vec<ToolSpec>> {
        // 从所有已连接 server 收集工具
        let mut all_tools = Vec::new();
        for server in &self.servers {
            for tool in &server.tools {
                all_tools.push(ToolSpec {
                    name: format!("{}__{}", server.name, tool.name),
                    description: tool.description.clone(),
                    input_schema: tool.input_schema.clone(),
                    source: ToolSource::MCP { server: server.name.clone() },
                    requires_confirmation: true,
                });
            }
        }
        Ok(all_tools)
    }
    
    async fn call(&self, name: &str, args: Value) -> Result<ToolCallOutcome> {
        let parts: Vec<&str> = name.splitn(2, "__").collect();
        if parts.len() != 2 {
            return Err(Error::InvalidToolName(name.to_string()));
        }
        
        let (server_name, tool_name) = (parts[0], parts[1]);
        let server = self.servers.iter()
            .find(|s| s.name == server_name)
            .ok_or(Error::ServerNotFound(server_name.to_string()))?;
        
        server.call_tool(tool_name, args).await
    }
}

6. 错误处理

rust
#[derive(Debug, thiserror::Error)]
pub enum McpError {
    #[error("server not connected: {0}")]
    NotConnected(String),
    
    #[error("server not found: {0}")]
    ServerNotFound(String),
    
    #[error("tool not found: {0}")]
    ToolNotFound(String),
    
    #[error("invalid tool name format: {0}")]
    InvalidToolName(String),
    
    #[error("call failed: {0}")]
    CallFailed(String),
    
    #[error("config error: {0}")]
    Config(#[from] toml::de::Error),
}

7. 使用示例

rust
use agent_mcp::{McpConfig, McpToolHost};
use agent_contract::ToolHost;

#[tokio::main]
async fn main() -> Result<()> {
    let config = McpConfig::load_default()?;
    let mut host = McpToolHost::from_config(config)?;
    host.connect_all().await?;
    
    let tools = host.list_tools().await?;
    println!("Available tools: {}", tools.len());
    
    let result = host.call("filesystem__read_file", json!({
        "path": "/workspace/README.md"
    })).await?;
    
    println!("Result: {}", result.content);
    Ok(())
}
