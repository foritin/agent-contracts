Tauri 应用壳

1. 概述

Tauri v2 应用壳，提供窗口管理、菜单、系统托盘、文件关联、自动更新等桌面应用基础能力。

3. 全局状态

rust
// hermes-gui/src/state.rs

pub struct AppState {
    pub config: Config,
    pub session_store: SessionStore,
    pub tool_host: Arc<RwLock<Box<dyn ToolHost>>>,
    pub current_session: RwLock<Option<String>>,
    pub event_sender: broadcast::Sender<AppEvent>,
}

impl AppState {
    pub async fn init() -> Result<Self> {
        let config = Config::load()?;
        
        // 初始化日志
        tracing_subscriber::fmt()
            .with_env_filter(&config.log_level)
            .init();
        
        // 创建存储目录
        tokio::fs::create_dir_all(&config.storage.base_dir).await?;
        
        // 初始化组件
        let session_store = SessionStore::new(config.storage.sessions_dir.clone());
        let tool_host = create_tool_host(&config).await?;
        let (event_sender, _) = broadcast::channel(256);
        
        Ok(Self {
            config,
            session_store,
            tool_host: Arc::new(RwLock::new(tool_host)),
            current_session: RwLock::new(None),
            event_sender,
        })
    }
}

4. Tauri 命令

rust
// hermes-gui/src/commands/chat.rs

#[tauri::command]
pub async fn send_message(
    state: State<'_, AppState>,
    message: String,
) -> Result<(), String> {
    let mut session_id = state.current_session.write().await;
    
    // 创建或恢复会话
    let session = if let Some(id) = session_id.as_ref() {
        state.session_store.load(id).await.map_err(|e| e.to_string())?
    } else {
        let session = state.session_store.create("claude-sonnet-4", "anthropic").await
            .map_err(|e| e.to_string())?;
        *session_id = Some(session.meta.id.clone());
        session
    };
    
    // 添加用户消息
    session.push_user(&message);
    state.session_store.append(&session.meta.id, SessionEvent::Message(
        Message::user_text(&message)
    )).await.map_err(|e| e.to_string())?;
    
    // 调用 LLM（在后台任务中）
    let tool_host = state.tool_host.read().await.clone();
    let event_sender = state.event_sender.clone();
    
    tokio::spawn(async move {
        match call_llm(&session, &*tool_host, event_sender).await {
            Ok(response) => {
                // 保存响应
            }
            Err(e) => {
                // 发送错误事件
                let _ = event_sender.send(AppEvent::Error(e.to_string()));
            }
        }
    });
    
    Ok(())
}

7. 使用示例

rust
// main.rs
fn main() {
    tracing_subscriber::fmt::init();
    
    let state = tokio::runtime::Builder::new_multi_thread()
        .enable_all()
        .build()
        .unwrap()
        .block_on(AppState::init())
        .expect("failed to initialize app state");
    
    tauri::Builder::default()
        .manage(state)
        .invoke_handler(tauri::generate_handler![
            commands::chat::send_message,
            commands::chat::cancel_stream,
            commands::session::list_sessions,
            commands::session::load_session,
            commands::session::delete_session,
            commands::config::get_config,
            commands::config::update_config,
            commands::mcp::list_mcp_tools,
        ])
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}
